/**
 * 只观察当前真实丘比特／混血儿窗口；固定原请求、同一 World 和字面付款／事件结果。
 * 临时修改仅限本 fixture 的请求字段、领域 current 和实例提交／事件 seam，finally 恢复。
 * 本实验不推导新 F02 政策、不调用未来 lease，也不把录制接受当领域付款。
 */
import { afterAll, expect, test } from "bun:test";
import { writeFileSync } from "node:fs";
import { bootstrapGame } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/app/bootstrap";
import { Camp, COMPONENT, Phase, Role, WinCondition, type ActionRequest, type GameEvent, type ToolCall } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/core";
import { CUPID_STAGE } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/mechanisms/roles/cupid/night_stage";
import { HYBRID_STAGE } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/mechanisms/roles/hybrid/night_stage";
import { getCupidState } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/mechanisms/roles/cupid/ports";
import { getHybridState } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/mechanisms/roles/hybrid/ports";
import { loversRecord } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/mechanisms/roles/cupid/relation";
import { GameActionRequestFactory } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/engine/action_request_factory";
import { nightStageContext } from "/home/hby/dev/AI-Werewolf-Arena/backend/tests/support/builders/night-stage-context";
import { sixPlayerMvpConfig } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/runtime/scenarios/six_player_mvp";
import { assertRequestDelivery, deferRequestDelivery, deriveBoundedRequest } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/core/domain/request_delivery";
import { ReproductionTrace } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/recording/reproduction-trace";
import { RULESET } from "/home/hby/dev/AI-Werewolf-Arena/packages/game-catalog/src/index";

const results: unknown[] = [];
afterAll(() => writeFileSync("/tmp/awa-cupid-hybrid-window-lease-baseline/results.json", JSON.stringify({ onlyBaseline: true, results }, null, 2) + "\n"));

for (const family of ["cupid", "hybrid"] as const) test(`${family}: original delivery observation, payment failure and late fact prefixes`, async () => {
  const runs: unknown[] = [];
  for (const mode of ["restored-input", "partial-payment", "late-fact"] as const) {
    const roles = family === "cupid"
      ? [Role.Cupid, Role.Hunter, Role.Wolf, Role.Villager, Role.Wolf, Role.Villager]
      : [Role.Hybrid, Role.Wolf, Role.HiddenWolf, Role.Seer, Role.Witch, Role.Villager];
    const config = { ...sixPlayerMvpConfig, boardSize: 6,
      roleSetups: [...new Set(roles)].map(role => ({ role, count: roles.filter(value => value === role).length })),
      winConditions: [WinCondition.SlaughterCity], enableSheriff: false, selfDestruct: { enabledWindows: [] },
      ...(family === "cupid" ? { cupid: { policy: "netease_2017_cupid_follows_lovers_v1" as const } } : {}),
    };
    const game = bootstrapGame(config, { fixedRoles: roles });
    const events: GameEvent[] = [];
    const role = game.world.getComponent(1, COMPONENT.Role)!;
    const state = family === "cupid" ? getCupidState(role)! : getHybridState(role)!;
    const initialVictory = game.world.getComponent(1, COMPONENT.Victory)!;
    const originalError = new Error(`${family}-${mode}`);
    const trace = new ReproductionTrace({ engine: { id: "awa-engine", sourceHash: "a".repeat(64), lockHash: "b".repeat(64) },
      random: game.random.metadata, ruleset: RULESET, boardId: `baseline_${family}`, normalizedBoard: game.boardConfig,
      locale: "zh-CN", maxDays: 1, dealMode: "fixed",
      players: roles.map((value, index) => ({ id: index + 1, name: `P${index + 1}`, role: value,
        camp: game.world.getComponent(index + 1, COMPONENT.Camp)!.camp })),
    }, () => {});
    const observations: string[] = [];
    let domainCurrent = true, calls = 0, notificationCount = 0;
    const observe = (label: string, request: ActionRequest) => {
      try { assertRequestDelivery(request); observations.push(`${label}:allowed`); }
      catch (error) { observations.push(`${label}:${error instanceof Error ? error.message : "unknown"}`); }
    };
    const originalAdd = game.world.addComponent;
    const originalPush = events.push;
    if (mode === "partial-payment") game.world.addComponent = function (id, component, value) {
      if (id === 1 && component === COMPONENT.Victory) throw originalError;
      return originalAdd.call(this, id, component, value);
    };
    if (mode === "late-fact") events.push = function (...items) {
      if (items.some(event => event.type === (family === "cupid" ? "cupid_lovers_chosen" : "hybrid_exemplar_chosen"))) throw originalError;
      return originalPush.apply(this, items);
    };
    const factory = new GameActionRequestFactory(game.world, events, 1);
    const ctx = nightStageContext({ world: game.world, events, isCurrent: () => domainCurrent,
      getAliveByRole: value => game.world.getAliveEntityIds().filter(id => game.world.getComponent(id, COMPONENT.Role)!.role === value),
      makeRequest: (actorId, allowedTools, extra) => factory.create({ actorId, allowedTools, phase: Phase.Night,
        stage: extra.stage, requiresAction: extra.requiresAction ?? true, summary: "Offline", context: extra }),
      actionProvider: { getAction: request => trace.decide(request, { getAction: async original => {
        calls++;
        if (mode === "restored-input") {
          const context = original.context, stage = context.stage;
          observe("original", original);
          (context as { stage: string }).stage = "baseline-invalid-stage";
          observe("bad-stage", original);
          (context as { stage: string }).stage = stage;
          observe("restored-stage", original);
          original.context = { ...context };
          observe("changed-context", original);
          original.context = context;
          observe("restored-context", original);
          domainCurrent = false;
          observe("bad-domain", original);
          domainCurrent = true;
          observe("restored-domain", original);
          const alias = deriveBoundedRequest(original, Date.now() + 60_000);
          observe("deadline-alias", alias);
          alias.context = { ...context };
          observe("alias-own-context", alias);
          observe("original-after-alias", original);
          observations.push(`deferred:${deferRequestDelivery(original, () => notificationCount++)}`);
        }
        return family === "cupid"
          ? { name: "choose_lovers", args: { first_id: 1, second_id: 3 } }
          : { name: "choose_exemplar", args: { target_id: 2 } } as ToolCall;
      } }) },
    });
    let caught: unknown;
    try { await (family === "cupid" ? CUPID_STAGE : HYBRID_STAGE).execute(ctx); }
    catch (error) { caught = error; }
    finally { game.world.addComponent = originalAdd; events.push = originalPush; }
    expect(calls).toBe(1);
    expect(trace.settlementCount).toBe(1);
    expect(state.remaining).toBe(0);
    expect(state.choice).toEqual(family === "cupid" ? { day: 1, pairIds: [1, 3] } : { day: 1, targetId: 2 });
    if (mode === "restored-input") {
      expect(caught).toBeUndefined();
      expect(observations).toEqual(["original:allowed", "bad-stage:allowed", "restored-stage:allowed", "changed-context:allowed", "restored-context:allowed", "bad-domain:allowed", "restored-domain:allowed", "deadline-alias:allowed", "alias-own-context:allowed", "original-after-alias:allowed", "deferred:false"]);
      expect(notificationCount).toBe(0);
    } else { expect(caught).toBe(originalError); expect(events.map(event => event.type)).toEqual([]); }
    if (mode === "partial-payment") expect(game.world.getComponent(1, COMPONENT.Victory)).toBe(initialVictory);
    else expect(game.world.getComponent(1, COMPONENT.Victory)!.affiliations).toEqual(family === "cupid" ? [{ kind: "relationship", groupId: "cupid_lovers" }] : [{ kind: "faction", faction: Camp.Wolf }]);
    if (family === "cupid") expect(loversRecord(game.world)?.pairIds).toEqual([1, 3]);
    runs.push({ mode, calls, observations, settlementCount: trace.settlementCount, remaining: state.remaining,
      choice: state.choice, sameError: mode === "restored-input" ? null : caught === originalError,
      events: events.map(event => event.type), originalVictoryRetained: game.world.getComponent(1, COMPONENT.Victory) === initialVictory });
  }
  results.push({ family, runs });
});
