import { expect, test } from "bun:test";
import { bootstrapGame } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/app/bootstrap";
import { COMPONENT } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/core/domain/components/names";
import { DeathCause, Phase, Role, type ActionRequest, type GameEvent, type RuntimeSnapshot } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/core/domain/model";
import { assertRequestDelivery, deferRequestDelivery, deriveBoundedRequest } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/core/domain/request_delivery";
import { GameActionRequestFactory } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/engine/action_request_factory";
import { ToolGateway } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/gateway/tool_gateway";
import { getDefaultLastWordsMechanism } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/mechanisms/last_words/last_words_mechanism";
import { LastWordsWindow } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/game/mechanisms/last_words/window";
import { sixPlayerMvpConfig } from "/home/hby/dev/AI-Werewolf-Arena/backend/src/runtime/scenarios/six_player_mvp";

function fixture(trace: string[]) {
  const game = bootstrapGame(sixPlayerMvpConfig, {
    fixedRoles: [Role.Wolf, Role.Wolf, Role.Seer, Role.Guard, Role.Villager, Role.Villager],
  });
  game.world.getComponent(5, COMPONENT.Alive)!.alive = false;
  const events: GameEvent[] = [];
  const state: RuntimeSnapshot = { day: 1, phase: Phase.Day, gameOver: false, result: null };
  const originalOpen = game.world.openDeathAction;
  let closeFailure: unknown;
  let shouldFailClose = false;
  game.world.openDeathAction = function(...args: Parameters<typeof originalOpen>) {
    trace.push("open");
    const permit = originalOpen.apply(this, args);
    return {
      id: permit.id,
      close() {
        trace.push("close");
        permit.close();
        if (shouldFailClose) throw closeFailure;
      },
    };
  };
  const window = new LastWordsWindow({
    requestFactory: GameActionRequestFactory,
    world: game.world,
    events,
    state,
    toolGateway: new ToolGateway(),
    lastWordsMechanism: getDefaultLastWordsMechanism(),
  });
  return {
    game, window, events,
    failClose(error: unknown) { closeFailure = error; shouldFailClose = true; },
  };
}

test("legacy last words have no envelope alias completion binding and preserve publication close failure prefixes", async () => {
  const trace: string[] = [], f = fixture(trace), notices: boolean[] = [];
  let original: ActionRequest | undefined;
  const push = f.events.push;
  f.events.push = function(...events: GameEvent[]) {
    trace.push("publish");
    return push.apply(this, events);
  };
  await f.window.collect([5], Phase.Day, Phase.Night, { 5: [DeathCause.WolfKill] }, {
    async getAction(request) {
      trace.push("provider");
      original = request;
      expect(request.context.stage).toBe("last_words");
      expect(request.context.day).toBe(1);
      expect(f.game.world.activeDeathActionCount).toBe(1);
      expect(deferRequestDelivery(request, accepted => notices.push(accepted))).toBe(false);
      const alias = deriveBoundedRequest(request, 0);
      expect(alias).not.toBe(request);
      expect(alias.context).toBe(request.context);
      alias.context = { ...alias.context, stage: "literal_bad_alias" as ActionRequest["context"]["stage"] };
      expect(() => assertRequestDelivery(alias)).not.toThrow();
      expect(() => assertRequestDelivery(request)).not.toThrow();
      expect(deferRequestDelivery(alias, accepted => notices.push(accepted))).toBe(false);
      request.actorId = 6;
      request.phase = Phase.Night;
      request.context.stage = "literal_bad_original" as ActionRequest["context"]["stage"];
      request.context.day = 99;
      request.deadlineAtMs = 0;
      expect(() => assertRequestDelivery(request)).not.toThrow();
      trace.push("return");
      return { name: "speak" as const, args: { text: " literal-baseline-words " } };
    },
  }, {});
  expect(trace).toEqual(["open", "provider", "return", "publish", "close"]);
  expect(original?.context.day).toBe(99);
  expect(notices).toEqual([]);
  expect(f.game.world.activeDeathActionCount).toBe(0);
  expect(f.events).toHaveLength(1);
  const event = f.events[0];
  expect(event.type).toBe("last_words_spoken");
  if (event.type !== "last_words_spoken") throw new Error("literal_missing_last_words");
  expect(event.payload).toEqual({ playerId: 5, phase: Phase.Day, day: 1, text: "literal-baseline-words" });

  const failureTrace: string[] = [], failed = fixture(failureTrace), failedNotices: boolean[] = [];
  const publicationFailure = new Error("literal_baseline_sink_failure");
  const closeFailure = new Error("literal_baseline_close_failure");
  failed.failClose(closeFailure);
  const originalPush = failed.events.push;
  failed.events.push = function(...events: GameEvent[]) {
    failureTrace.push("append-then-throw");
    originalPush.apply(this, events);
    throw publicationFailure;
  };
  await expect(failed.window.collect([5], Phase.Day, Phase.Night, { 5: [DeathCause.WolfKill] }, {
    async getAction(request) {
      failureTrace.push("provider");
      expect(deferRequestDelivery(request, accepted => failedNotices.push(accepted))).toBe(false);
      return { name: "speak" as const, args: { text: "literal-retained-prefix" } };
    },
  }, {})).rejects.toBe(closeFailure);
  expect(failureTrace).toEqual(["open", "provider", "append-then-throw", "close"]);
  expect(failed.events).toHaveLength(1);
  expect(failed.events[0].type).toBe("last_words_spoken");
  expect(failedNotices).toEqual([]);
  expect(failed.game.world.activeDeathActionCount).toBe(0);
  console.log(JSON.stringify({
    characterization: "legacy-last-words-delivery",
    traces: [trace, failureTrace],
    unboundOriginalAndAlias: true,
    deferReturns: [false, false, false],
    originalEnvelopeMutationStillPublishesCapturedActor: 5,
    notices: [notices, failedNotices],
    failureWinner: "literal_baseline_close_failure",
    appendPrefixRetained: failed.events.length,
  }));
});
