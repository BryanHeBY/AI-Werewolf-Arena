type Mode = "envelope" | "first-data-write" | "fact-sink";
type Row = Record<string, unknown>;
const results: Row[] = [];
const resultPath = "/tmp/awa-q2-ordinary-delivery/results.json";
function save() {
  writeFileSync(resultPath, JSON.stringify({
    policy: "Current behavior only; action outcome alone is not classified as a bug.",
    fixtures: results,
    limits: [
      "Two declarations / six fresh fixtures, not six independent test declarations.",
      "Actual NightPipeline and DeathResolution use real factory/gateway/domain owners; no manual delivery guard/completion is installed.",
      "Trace fixture engine hashes are explicit schema-only placeholders; this is not a strict replay/archive build identity check.",
      "Source provenance is frozen separately; transitive modules are the Root execution workspace, not an asserted complete fixed Git closure.",
    ],
  }, null, 2) + "\n");
}
function errorData(error: unknown) {
  return error instanceof Error ? { name: error.name, message: error.message } : { thrown: String(error) };
}
function delivery(label: string, request: ActionRequest): Row {
  try { assertRequestDelivery(request); return { label, returned: true }; }
  catch (error) { return { label, returned: false, error: errorData(error) }; }
}
function observeEnvelope(request: ActionRequest): Row[] {
  const observations = [delivery("original-before", request)];
  const originalContext = request.context;
  const alias = deriveBoundedRequest(request, Date.now() + 30_000);
  observations.push({ label: "bounded-alias-produced", sameObject: alias === request, sameContext: alias.context === originalContext });
  // 只改变本次原请求的 context 身份；不改 World、生命、库存或领域 current 回调。
  request.context = { ...originalContext };
  observations.push(delivery("original-context-replaced", request));
  request.context = originalContext;
  observations.push(delivery("original-context-restored", request));
  // 当前普通窗口是否登记传播不能从 derive 返回对象推定；只记录实际 delivery 观察。
  observations.push(delivery("bounded-alias-after-original-restore", alias));
  return observations;
}
function base(kind: "seer" | "badge", mode: Mode) {
  const roles = [Role.Wolf, Role.Wolf, Role.Seer, Role.Guard, Role.Villager, Role.Villager];
  const config = kind === "badge" ? {
    ...sixPlayerMvpConfig,
    enableSheriff: true,
    initialSheriffSeat: 5,
    sheriff: { voteWeight: 2.25, badgePolicy: "awa_holder_choice_v1" as const },
  } : sixPlayerMvpConfig;
  const game = bootstrapGame(config, { fixedRoles: roles, random: createRandom(`q2-${kind}-${mode}`), locale: "en" });
  const events: GameEvent[] = [];
  const chronology: Row[] = [];
  const notifications: Row[] = [];
  const snapshot: RuntimeSnapshot = { day: kind === "badge" ? 2 : 1, phase: kind === "badge" ? Phase.Voting : Phase.Night, gameOver: false, result: null };
  let trace!: ReproductionTrace;
  trace = new ReproductionTrace({
    engine: { id: "awa-engine", sourceHash: "a".repeat(64), lockHash: "b".repeat(64) },
    random: game.random.metadata, ruleset: RULESET, boardId: `q2_${kind}_${mode.replaceAll("-", "_")}`,
    normalizedBoard: structuredClone(game.boardConfig), locale: "en", maxDays: 2, dealMode: "fixed",
    players: game.world.entityIds().map(id => ({
      id,
      name: game.world.getComponent(id, COMPONENT.Identity)!.name,
      role: game.world.getComponent(id, COMPONENT.Role)!.role,
      camp: game.world.getComponent(id, COMPONENT.Camp)!.camp,
    })),
  }, () => {
    notifications.push({ decisions: trace.decisionCount, settlements: trace.settlementCount });
    chronology.push({ point: "trace-onChange", decisions: trace.decisionCount, settlements: trace.settlementCount });
  }, () => events);
  trace.checkpoint(captureCheckpoint(game.world, snapshot, 0, true, 0), []);
  const injected = new Error(`literal-q2-${kind}-${mode}`);
  let captured: ActionRequest | undefined;
  let envelope: Row[] = [];
  let restoreWrite: (() => void) | undefined;
  let writeAttempts = 0;
  let sinkAttempts = 0;
  const pushDescriptor = Object.getOwnPropertyDescriptor(events, "push");
  const originalPush = events.push;
  if (mode === "fact-sink") {
    events.push = function(...rows: GameEvent[]) {
      if (rows.some(row => row.type === (kind === "seer" ? "seer_checked" : "sheriff_badge_transferred"))) {
        sinkAttempts++;
        chronology.push({ point: "fact-sink-before-append", settlements: trace.settlementCount, existingEvents: this.length });
        throw injected;
      }
      return originalPush.apply(this, rows);
    };
  }
  const candidate: ToolCall = kind === "seer" ? { name: "check_identity", args: { target_id: 1 } }
    : { name: "choose_badge_successor", args: { target_id: 6 } };
  const targetStage = kind === "seer" ? "seer" : "sheriff_badge_choice";
  const provider: ActionProvider = {
    async getAction(request) {
      if (request.context.stage !== targetStage) return null;
      captured = request;
      chronology.push({ point: "actual-window-provider", actorId: request.actorId, day: request.context.day, stage: request.context.stage, permitCount: game.world.activeDeathActionCount });
      return trace.decide(request, {
        async getAction(original) {
          expect(original).toBe(request);
          expect(request.actorId).toBe(kind === "seer" ? 3 : 5);
          if (kind === "badge") {
            expect(game.world.getComponent(5, COMPONENT.Alive)!.alive).toBe(false);
            expect(game.world.hasDeathAction(5, "choose_badge_successor", Phase.Voting, 2, request.context.death_action_id)).toBe(true);
          }
          if (mode === "envelope") envelope = observeEnvelope(request);
          if (mode === "first-data-write") {
            const object = kind === "seer" ? getSeerState(game.world.getComponent(3, COMPONENT.Role)!)!
              : game.world.getComponent(5, COMPONENT.Badge)!;
            const key = kind === "seer" ? "lastTarget" : "isSheriff";
            const descriptor = Object.getOwnPropertyDescriptor(object, key);
            if (!descriptor || !("value" in descriptor)) throw new Error("q2_original_data_field_missing");
            Object.defineProperty(object, key, {
              configurable: true, enumerable: descriptor.enumerable,
              get() { return descriptor.value; },
              set(value: unknown) {
                writeAttempts++;
                chronology.push({ point: "first-owned-data-write", key, attemptedValue: value, settlements: trace.settlementCount });
                throw injected;
              },
            });
            restoreWrite = () => Object.defineProperty(object, key, descriptor);
          }
          chronology.push({ point: "provider-candidate-return", settlements: trace.settlementCount });
          return candidate;
        },
      });
    },
  };
  return { game, events, chronology, notifications, snapshot, trace, provider, injected,
    restore() { restoreWrite?.(); if (pushDescriptor) Object.defineProperty(events, "push", pushDescriptor); else delete (events as { push?: unknown }).push; },
    observations() { return { captured, envelope, writeAttempts, sinkAttempts }; },
  };
}
async function run(kind: "seer" | "badge", mode: Mode) {
  const f = base(kind, mode);
  const row: Row = { kind, mode, reachedWindow: false };
  let error: unknown, threw = false;
  try {
    if (kind === "seer") {
      await new NightPipeline({ world: f.game.world, events: f.events, toolGateway: new ToolGateway(), damageResolutionSystem: new DamageResolutionSystem(), random: f.game.random, rulePlan: f.game.rulePlan }).execute(f.game.boardConfig, f.provider, 1);
    } else {
      await new DeathResolution(f.game.world, f.events, new ToolGateway(), new EventRegistry(), f.snapshot, f.game.boardConfig).resolve([5], { 5: [DeathCause.Exile] }, f.provider, Phase.Voting, Phase.Voting);
    }
  } catch (caught) { threw = true; error = caught; }
  finally { f.restore(); }
  const observed = f.observations();
  Object.assign(row, { reachedWindow: observed.captured !== undefined,
    thrown: threw, sameInjectedError: threw && error === f.injected, error: threw ? errorData(error) : null,
    writeAttempts: observed.writeAttempts, sinkAttempts: observed.sinkAttempts,
    envelope: observed.envelope, chronology: f.chronology, notifications: f.notifications,
    decisionCount: f.trace.decisionCount, settlementCount: f.trace.settlementCount,
    events: f.events.map(({ type, payload, audience }) => ({ type, payload, ...(audience ? { audience } : {}) })),
    afterWindowDelivery: observed.captured ? delivery("original-after-window", observed.captured) : null,
    activeDeathActionCountAfterFinally: f.game.world.activeDeathActionCount,
  });
  if (kind === "seer") {
    const role = f.game.world.getComponent(3, COMPONENT.Role)!;
    const ability = findNightAbility(role, "seer_check")!;
    Object.assign(row, { seer: { abilityId: ability.id, lastUsedDay: ability.lastUsedDay, uses: ability.uses, disabled: ability.disabled, state: structuredClone(getSeerState(role)!) } });
  } else {
    Object.assign(row, { badge: {
      fromAlive: f.game.world.getComponent(5, COMPONENT.Alive)!.alive,
      from: { ...f.game.world.getComponent(5, COMPONENT.Badge)! },
      to: { ...f.game.world.getComponent(6, COMPONENT.Badge)! },
      fromVoting: { ...f.game.world.getComponent(5, COMPONENT.VotingRight)! },
      toVoting: { ...f.game.world.getComponent(6, COMPONENT.VotingRight)! },
      oldPermitCurrentAfterFinally: observed.captured ? f.game.world.hasDeathAction(5, "choose_badge_successor", Phase.Voting, 2, observed.captured.context.death_action_id) : null,
    } });
  }
  try { row.recordedDecisions = f.trace.build().decisions; }
  catch (caught) { row.traceBuildError = errorData(caught); }
  results.push(row); save();
  // 只核真实fixture到达及注入同Error，不预置action/error/pending、知识/徽或事件应有何政策。
  expect(row.reachedWindow).toBe(true);
  if (kind === "badge") expect(row.activeDeathActionCountAfterFinally).toBe(0);
  if (mode !== "envelope") {
    expect(row.sameInjectedError).toBe(true);
    expect(mode === "first-data-write" ? row.writeAttempts : row.sinkAttempts).toBe(1);
  } else expect(row.thrown).toBe(false);
}
for (const kind of ["seer", "badge"] as const)
  test(`Q2 current ${kind} envelope and actual data/fact failure prefixes`, async () => {
    for (const mode of ["envelope", "first-data-write", "fact-sink"] as const) await run(kind, mode);
  });
